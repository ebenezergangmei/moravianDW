<script>
(function () {
  'use strict';

  var ROOT = document.getElementById('moravian-watchword-app');
  if (!ROOT) return;

  /* ---------- SETTINGS ---------- */
  var PAGES_BASE = 'https://ebenezergangmei.github.io/moravianDW/';
  var RAW_BASE = 'https://raw.githubusercontent.com/ebenezergangmei/moravianDW/main/';
  // Tries GitHub Pages first, then the raw GitHub address as a backup.
  var DATA_BASES = [PAGES_BASE + 'data/', RAW_BASE + 'data/'];
  // true  = hide the normal Blogger posts so this page shows only the app.
  // false = leave the rest of the blog untouched.
  var HIDE_BLOG_POSTS = true;
  /* ------------------------------- */

  if (HIDE_BLOG_POSTS && document.body) document.body.classList.add('mw-app-only');

  // Safe storage: never crashes if the browser blocks storage.
  var store = {
    get: function (k) { try { return localStorage.getItem(k); } catch (e) { return null; } },
    set: function (k, v) { try { localStorage.setItem(k, v); } catch (e) {} }
  };

  function $(id) { return document.getElementById(id); }
  var languageSelect = $('mw-language'), yearSelect = $('mw-year');
  var previousBtn = $('mw-prev'), todayBtn = $('mw-today'), nextBtn = $('mw-next');
  var calendarBtn = $('mw-calendar-btn'), calendarPanel = $('mw-calendar-panel');
  var calendarPrev = $('mw-cal-prev'), calendarNext = $('mw-cal-next');
  var calendarTitle = $('mw-cal-title'), calendarDays = $('mw-cal-days'), calendarToday = $('mw-cal-today');
  var fontDown = $('mw-font-down'), fontUp = $('mw-font-up'), fontLabel = $('mw-font-size');
  var howBtn = $('mw-how'), statusEl = $('mw-status'), contentEl = $('mw-content');
  var dialog = $('mw-dialog'), dialogClose = $('mw-dialog-close');

  var STORAGE_LANGUAGE = 'moravianLanguage';
  var STORAGE_FONT = 'moravianFontScale';
  var CACHE_LIST = 'moravianGitHubFileListV1';
  var CACHE_PREFIX = 'moravianGitHubTxtV1:';
  var MIN_FONT = 80, MAX_FONT = 140, FONT_STEP = 10;

  var availableFiles = [];
  var selectedLanguage = '';
  var selectedYear = 0;
  var records = [];
  var currentDate = new Date();
  var calendarMonth = currentDate.getMonth();
  var calendarYear = currentDate.getFullYear();
  var fontScale = Number(store.get(STORAGE_FONT)) || 100;
  var offlineNotice = false;

  /* ---------- helpers ---------- */
  function status(msg) { statusEl.textContent = msg || ''; }
  function errorBox(msg) {
    contentEl.innerHTML = '';
    var d = document.createElement('div');
    d.className = 'mw-error';
    d.textContent = msg;
    contentEl.appendChild(d);
  }
  function daysInYear(y) { return new Date(y, 1, 29).getMonth() === 1 ? 366 : 365; }
  function dayOfYear(d) {
    var start = new Date(d.getFullYear(), 0, 1);
    var cur = new Date(d.getFullYear(), d.getMonth(), d.getDate());
    return Math.round((cur - start) / 86400000) + 1;
  }
  function formatDate(d) {
    return d.toLocaleDateString(undefined, { weekday: 'long', year: 'numeric', month: 'long', day: 'numeric' });
  }
  function sameDay(a, b) {
    return a.getFullYear() === b.getFullYear() && a.getMonth() === b.getMonth() && a.getDate() === b.getDate();
  }

  // Each "#" line starts one day. Section 1 = January 1.
  function parseText(text) {
    var normalized = String(text).replace(/\r\n/g, '\n').replace(/\r/g, '\n');
    return normalized.split(/^\s*#/m).slice(1).map(function (block) {
      var lines = block.split('\n');
      while (lines.length && lines[0].trim() === '') lines.shift();
      while (lines.length && lines[lines.length - 1].trim() === '') lines.pop();
      return { lines: lines };
    }).filter(function (r) { return r.lines.length > 0; });
  }

  // File names look like "2027 Hindi.txt"
  function parseFileInfo(name) {
    var m = String(name).match(/^(\d{4})\s+(.+?)\.txt$/i);
    return m ? { name: String(name), year: Number(m[1]), language: m[2].trim() } : null;
  }
  function getLanguages() {
    var a = [];
    availableFiles.forEach(function (f) {
      var i = parseFileInfo(f);
      if (i && a.indexOf(i.language) < 0) a.push(i.language);
    });
    return a.sort(function (x, y) { return x.localeCompare(y); });
  }
  function getYears(lang) {
    var a = [];
    availableFiles.forEach(function (f) {
      var i = parseFileInfo(f);
      if (i && i.language === lang && a.indexOf(i.year) < 0) a.push(i.year);
    });
    return a.sort(function (x, y) { return x - y; });
  }
  function findFile(lang, year) {
    for (var i = 0; i < availableFiles.length; i++) {
      var x = parseFileInfo(availableFiles[i]);
      if (x && x.language === lang && x.year === year) return x.name;
    }
    return null;
  }

  /* ---------- loading ---------- */
  // Reads a data file, trying each web address in turn.
  async function fetchData(name) {
    var lastError = null;
    for (var i = 0; i < DATA_BASES.length; i++) {
      var url = DATA_BASES[i] + name;
      try {
        var r = await fetch(url + '?v=' + Date.now(), { cache: 'no-store' });
        if (!r.ok) throw new Error('HTTP ' + r.status + ' for ' + url);
        return await r.text();
      } catch (e) { lastError = e; }
    }
    throw lastError || new Error('Could not read ' + name);
  }

  async function loadFileList() {
    try {
      var data = JSON.parse(await fetchData('files.json'));
      if (!Array.isArray(data)) throw new Error('data/files.json is not a list.');
      availableFiles = data.map(function (x) {
        return typeof x === 'string' ? x : (x && x.name ? x.name : null);
      }).filter(Boolean);
      store.set(CACHE_LIST, JSON.stringify(availableFiles));
    } catch (e) {
      var cached = store.get(CACHE_LIST);
      if (cached) {
        try {
          availableFiles = JSON.parse(cached);
          offlineNotice = true;
          status('Could not reach GitHub. Using the last saved file list.');
          return;
        } catch (_) {}
      }
      throw new Error('Cannot read the Watchword file list.\n\n' + (e && e.message ? e.message : e));
    }
  }

  async function loadSelectedFile() {
    var fileName = findFile(selectedLanguage, selectedYear);
    if (!fileName) {
      records = [];
      errorBox('No Watchword file found for ' + selectedLanguage + ' ' + selectedYear + '.');
      updateButtons();
      return false;
    }
    status('Reading ' + fileName + '...');
    try {
      var text = null;
      try {
        text = await fetchData(encodeURIComponent(fileName));
        store.set(CACHE_PREFIX + fileName, text); // failure to save is ignored
        offlineNotice = false;
      } catch (networkError) {
        text = store.get(CACHE_PREFIX + fileName);
        if (text === null) throw networkError;
        offlineNotice = true;
        status('Could not reach GitHub. Showing the saved copy of ' + fileName + '.');
      }
      var parsed = parseText(text);
      var expected = daysInYear(selectedYear);
      if (parsed.length !== expected) {
        throw new Error('The file ' + fileName + ' has ' + parsed.length + ' # sections, but ' + selectedYear + ' needs ' + expected + '.');
      }
      records = parsed;
      if (currentDate.getFullYear() !== selectedYear) currentDate = dateForYear();
      calendarYear = selectedYear;
      calendarMonth = currentDate.getMonth();
      renderCurrent();
      renderCalendar();
      updateButtons();
      if (!offlineNotice) status('');
      return true;
    } catch (e) {
      records = [];
      errorBox('Could not load the Watchwords:\n\n' + (e && e.message ? e.message : e) + '\n\nIf this is your first visit, please make sure you are online.');
      status('');
      updateButtons();
      return false;
    }
  }

  /* ---------- selectors ---------- */
  function populateLanguages() {
    var langs = getLanguages();
    languageSelect.innerHTML = '';
    langs.forEach(function (l) {
      var o = document.createElement('option');
      o.value = l; o.textContent = l;
      languageSelect.appendChild(o);
    });
    if (!langs.length) {
      languageSelect.innerHTML = '<option value="">No languages</option>';
      selectedLanguage = '';
      return;
    }
    var saved = store.get(STORAGE_LANGUAGE);
    selectedLanguage = (saved && langs.indexOf(saved) >= 0) ? saved : langs[0];
    languageSelect.value = selectedLanguage;
  }

  // keep = year to stay on when it exists for this language.
  // Otherwise: this year if available, else the nearest year.
  function populateYears(keep) {
    var years = getYears(selectedLanguage);
    yearSelect.innerHTML = '';
    years.forEach(function (y) {
      var o = document.createElement('option');
      o.value = String(y); o.textContent = String(y);
      yearSelect.appendChild(o);
    });
    if (!years.length) {
      yearSelect.innerHTML = '<option value="">No years</option>';
      selectedYear = 0;
      return;
    }
    var thisYear = new Date().getFullYear();
    if (keep && years.indexOf(keep) >= 0) selectedYear = keep;
    else if (years.indexOf(thisYear) >= 0) selectedYear = thisYear;
    else {
      selectedYear = years.reduce(function (best, y) {
        var dy = Math.abs(y - thisYear), db = Math.abs(best - thisYear);
        return (dy < db || (dy === db && y > best)) ? y : best;
      }, years[0]);
    }
    yearSelect.value = String(selectedYear);
  }

  // Today's month and day, placed inside the selected year.
  // Example: today is Oct 2 and the year is 2027 -> Oct 2, 2027.
  // Feb 29 in a year with no Feb 29 becomes Feb 28.
  function dateForYear() {
    var t = new Date(), m = t.getMonth(), d = t.getDate();
    var c = new Date(selectedYear, m, d);
    if (c.getMonth() !== m) c = new Date(selectedYear, m + 1, 0);
    return c;
  }

  // Used when the language or year changes: always show today's date.
  function keepDate() {
    currentDate = dateForYear();
    calendarYear = selectedYear;
    calendarMonth = currentDate.getMonth();
  }

  /* ---------- display ---------- */
  function getCurrentRecord() {
    if (!records.length || currentDate.getFullYear() !== selectedYear) return null;
    var i = dayOfYear(currentDate) - 1;
    return (i >= 0 && i < records.length) ? records[i] : null;
  }
  function renderCurrent() {
    contentEl.innerHTML = '';
    var rec = getCurrentRecord();
    if (!rec) { errorBox('No Watchword record found for ' + formatDate(currentDate) + '.'); return; }
    var entry = document.createElement('div');
    entry.className = 'mw-entry';
    rec.lines.forEach(function (line) {
      var d = document.createElement('div');
      d.className = 'mw-line';
      d.textContent = line;
      entry.appendChild(d);
    });
    contentEl.appendChild(entry);
  }
  function updateButtons() {
    if (!records.length) { previousBtn.disabled = true; nextBtn.disabled = true; return; }
    var d = dayOfYear(currentDate), t = daysInYear(selectedYear);
    previousBtn.disabled = d <= 1;
    nextBtn.disabled = d >= t;
  }
  function refresh() { renderCurrent(); renderCalendar(); updateButtons(); }

  function goPrev() {
    if (dayOfYear(currentDate) <= 1) return;
    currentDate = new Date(currentDate.getFullYear(), currentDate.getMonth(), currentDate.getDate() - 1);
    calendarMonth = currentDate.getMonth(); refresh();
  }
  function goNext() {
    if (dayOfYear(currentDate) >= daysInYear(selectedYear)) return;
    currentDate = new Date(currentDate.getFullYear(), currentDate.getMonth(), currentDate.getDate() + 1);
    calendarMonth = currentDate.getMonth(); refresh();
  }
  function goToday() {
    currentDate = dateForYear();
    calendarYear = selectedYear; calendarMonth = currentDate.getMonth(); refresh();
  }

  /* ---------- calendar ---------- */
  function renderCalendar() {
    calendarTitle.textContent = new Date(calendarYear, calendarMonth, 1)
      .toLocaleDateString(undefined, { month: 'long', year: 'numeric' });
    calendarPrev.disabled = calendarMonth <= 0;
    calendarNext.disabled = calendarMonth >= 11;
    calendarDays.innerHTML = '';
    var first = new Date(calendarYear, calendarMonth, 1).getDay();
    var total = new Date(calendarYear, calendarMonth + 1, 0).getDate();
    var now = new Date();
    for (var i = 0; i < first; i++) {
      var e = document.createElement('div');
      e.className = 'mw-cal-empty';
      calendarDays.appendChild(e);
    }
    for (var day = 1; day <= total; day++) {
      var b = document.createElement('button');
      b.type = 'button';
      b.textContent = String(day);
      var dt = new Date(calendarYear, calendarMonth, day);
      if (sameDay(dt, currentDate)) b.classList.add('mw-selected');
      if (sameDay(dt, now)) b.classList.add('mw-today-mark');
      b.addEventListener('click', (function (x) {
        return function () { currentDate = x; refresh(); closeCalendar(); };
      })(dt));
      calendarDays.appendChild(b);
    }
  }
  function openCalendar() {
    calendarPanel.hidden = false;
    calendarBtn.setAttribute('aria-expanded', 'true');
    calendarYear = selectedYear; calendarMonth = currentDate.getMonth();
    renderCalendar();
  }
  function closeCalendar() {
    calendarPanel.hidden = true;
    calendarBtn.setAttribute('aria-expanded', 'false');
  }
  function changeCalendarMonth(delta) {
    var m = calendarMonth + delta;
    if (m < 0 || m > 11) return;
    calendarMonth = m; renderCalendar();
  }

  /* ---------- font size ---------- */
  function applyFont() {
    fontScale = Math.max(MIN_FONT, Math.min(MAX_FONT, fontScale));
    ROOT.style.setProperty('--mw-reader-size', (18 * fontScale / 100) + 'px');
    fontLabel.textContent = fontScale + '%';
    store.set(STORAGE_FONT, String(fontScale));
  }

  /* ---------- events ---------- */
  languageSelect.addEventListener('change', async function () {
    selectedLanguage = languageSelect.value;
    store.set(STORAGE_LANGUAGE, selectedLanguage);
    populateYears(selectedYear);
    keepDate();
    await loadSelectedFile();
  });
  yearSelect.addEventListener('change', async function () {
    selectedYear = Number(yearSelect.value);
    keepDate();
    await loadSelectedFile();
  });
  previousBtn.addEventListener('click', goPrev);
  nextBtn.addEventListener('click', goNext);
  todayBtn.addEventListener('click', goToday);
  calendarBtn.addEventListener('click', function () { if (calendarPanel.hidden) openCalendar(); else closeCalendar(); });
  calendarPrev.addEventListener('click', function () { changeCalendarMonth(-1); });
  calendarNext.addEventListener('click', function () { changeCalendarMonth(1); });
  calendarToday.addEventListener('click', function () { goToday(); closeCalendar(); });
  document.addEventListener('click', function (e) {
    if (!calendarPanel.hidden && !e.target.closest('.mw-calendar-wrap')) closeCalendar();
  });
  fontDown.addEventListener('click', function () { fontScale -= FONT_STEP; applyFont(); });
  fontUp.addEventListener('click', function () { fontScale += FONT_STEP; applyFont(); });
  howBtn.addEventListener('click', function () { dialog.classList.add('open'); });
  dialogClose.addEventListener('click', function () { dialog.classList.remove('open'); });
  dialog.addEventListener('click', function (e) { if (e.target === dialog) dialog.classList.remove('open'); });
  document.addEventListener('keydown', function (e) {
    if (e.key === 'Escape') { dialog.classList.remove('open'); closeCalendar(); }
  });

  /* ---------- start ---------- */
  async function start() {
    try {
      applyFont();
      status('Reading the Watchword list...');
      await loadFileList();
      populateLanguages();
      populateYears();
      if (!selectedLanguage || !selectedYear) throw new Error('No valid language or year was found in data/files.json.');
      currentDate = dateForYear();
      calendarYear = selectedYear;
      calendarMonth = currentDate.getMonth();
      await loadSelectedFile();
    } catch (e) {
      errorBox('Could not start the Watchword app.\n\n' + (e && e.message ? e.message : e));
      status('');
    }
  }
  start();
})();
